cmake_minimum_required(VERSION 3.24)
project(RomajiAligner LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET 15.0 CACHE STRING "Minimum macOS version")
set(QWEN3_ASR_TEST OFF CACHE BOOL "" FORCE)
set(QWEN3_ASR_TIMING OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../.cache/aligner-source")
if(NOT EXISTS "${DEPS}/qwen3-asr.cpp/CMakeLists.txt")
    message(FATAL_ERROR "Run python3 scripts/fetch-aligner-source.py once; checks never fetch dependencies.")
endif()
file(READ "${DEPS}/qwen3-asr.cpp/SOURCE.sha" QWEN_PIN)
file(READ "${DEPS}/qwen3-asr.cpp/ggml/SOURCE.sha" GGML_PIN)
string(STRIP "${QWEN_PIN}" QWEN_PIN)
string(STRIP "${GGML_PIN}" GGML_PIN)
if(NOT QWEN_PIN STREQUAL "6dcc586e5073fd6e85ee5728e75f0903d6c70c6c" OR NOT GGML_PIN STREQUAL "9be313313c8ecb9488911bd64550190e3ed80f38")
    message(FATAL_ERROR "Alignment source pins differ; rerun the explicit source bootstrap.")
endif()
set(ENV{GIT_CEILING_DIRECTORIES} "${DEPS}/qwen3-asr.cpp")
add_subdirectory("${DEPS}/qwen3-asr.cpp" upstream EXCLUDE_FROM_ALL)
add_executable(RomajiAligner main.cpp)
target_link_libraries(RomajiAligner PRIVATE qwen3-asr)
target_compile_options(RomajiAligner PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(RomajiAligner PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
